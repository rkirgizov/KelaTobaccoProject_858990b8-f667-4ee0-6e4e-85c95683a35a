//REGION FILL_USE_PIPE
// 
// -----------------------------------------------------------------------------------------
// FILL PIPE 
// -----------------------------------------------------------------------------------------
//
// known blend
IF
Combined(_TobaccoPouch,_SmokingPipe,_,_,_,_Player,_EmptyPouch)
AND
NOT DB_KTP_UnknownBlendPouch(_TobaccoPouch,_,_)
AND
GetTemplate(_TobaccoPouch,_TobaccoPouchTemplate)
AND
DB_KTP_StatusPouch(_StatusTurns,_PuffCount,_Status,_TobaccoPouchTemplate)
AND
GetStatString(_SmokingPipe,_SmokingPipeStats)
AND
_SmokingPipeStats == "OBJ_Smoking_Pipe"
THEN
//SetDualEntityEvent(_SmokingPipe,_SmokingPipe,"FILL PIPE");
SetTag(_SmokingPipe,KTP_FILLED_d0575f41-9d81-4f26-9f34-530c2162c110); // for condition KTPHasTobaccoStatus()
DB_KTP_FilledPipes(_SmokingPipe,_Status,_PuffCount,_StatusTurns);
PROC_KTP_RemoveOrLeaveFakePouchTobacco(_Player,_EmptyPouch); 
// 
// unknown blend
IF
Combined(_TobaccoPouchUnknown,_SmokingPipe,_,_,_,_Player,_EmptyPouch)
AND
DB_KTP_UnknownBlendPouch(_TobaccoPouchUnknown,_TobaccoPouchUnknownTemplate,_TobaccoPouchUnknownStatusPattern)
AND
GetStatString(_SmokingPipe,_SmokingPipeStats)
AND
_SmokingPipeStats == "OBJ_Smoking_Pipe"
THEN
//SetDualEntityEvent(_SmokingPipe,_SmokingPipe,"FILL PIPE WITH UNKNOWN BLEND");
SetTag(_SmokingPipe,KTP_FILLED_d0575f41-9d81-4f26-9f34-530c2162c110); // for condition KTPHasTobaccoStatus()
DB_KTP_FilledPipesUnknownBlend(_SmokingPipe,_TobaccoPouchUnknownStatusPattern);
NOT DB_KTP_UnknownBlendPouch(_TobaccoPouchUnknown,_TobaccoPouchUnknownTemplate,_TobaccoPouchUnknownStatusPattern);
PROC_KTP_RemoveOrLeaveFakePouchTobacco(_Player,_EmptyPouch); 
// 
// If there is no empty pouch in the inventory, then we leave a new one, if there is, we delete it
PROC
PROC_KTP_RemoveOrLeaveFakePouchTobacco((CHARACTER)_Player,(ITEM)_EmptyPouch)
AND
GetTemplate(_EmptyPouch,_EmptyPouchTemplate)
AND
TemplateIsInInventory((ITEMROOT)_EmptyPouchTemplate,_Player,_Count)
AND
_Count > 0
THEN
ObjectTimerLaunch(_EmptyPouch,"TIMER_KTP_DeleteFakePouchTobacco",30);
//
// After a long rest, we remove all tobacco statuses
PROC
PROC_LongRest()
AND
DB_Players(_Character)
AND
DB_KTP_StatusPouch(_,_,_TobaccoStatus,_)
THEN
RemoveStatus(_Character,_TobaccoStatus);
// 
//
// -----------------------------------------------------------------------------------------
// USE PIPE 
// -----------------------------------------------------------------------------------------
//
// Remove old status
IF
UseStarted(_Player, _SmokingPipe)
AND
DB_KTP_FilledPipes(_SmokingPipe,_,_,_)
AND
DB_KTP_StatusPouch(_,_,_TobaccoStatus,_)
AND
HasActiveStatus(_Player,_TobaccoStatus,1)
THEN
RemoveStatus(_Player,_TobaccoStatus);
//
// known blend
IF
UseFinished(_Player,_SmokingPipe,1)
AND
DB_KTP_FilledPipes(_SmokingPipe,_Status,_PuffCount,_StatusTurns)
AND
DB_KTP_StatusPouch(_,_,_Status,_TobaccoPouchTemplate)
AND
NOT HasActiveStatus(_SmokingPipe,"KTP_UNKNOWN",1)
AND
_PuffCount > 0
AND
IntegerSubtract(_PuffCount,1,_NewPuffCount)
AND
NOT HasActiveStatus(_Player,"KTP_OVERDOSE_5",1)
AND
QRY_KTP_CheckHalsinPipe(_Player,_SmokingPipe,_Status)
AND
QRY_KTP_CheckForOverdose(_Player)
//AND
//Concatenate("USE PIPE KNOWN BLEND - ",_Status,_StringDebug)
THEN
//SetDualEntityEvent(_SmokingPipe,_SmokingPipe,_StringDebug);
PlayEffect(_Player,(EFFECTRESOURCE)VFX_Status_Tharchiate_False_Life_Apply_01_e34f6123-0933-e70d-339e-d5a8c3490cb0,"Chest_M",1.0);
PlaySound(_Player, "GOB_GoblinKing_Scrying_SmokeStart");
ApplyStatus(_Player,_Status,_StatusTurns,1);
DB_KTP_FilledPipes(_SmokingPipe,_Status,_NewPuffCount,_StatusTurns);
PROC_StudyIngredientsOfTobaccoBlend((STRING)_Status,(ITEMROOT)_TobaccoPouchTemplate); // study recipe 
//
// USE PIPE UNKNOWN BLEND
IF
UseFinished(_Player,_SmokingPipe,1)
AND
HasActiveStatus(_SmokingPipe,"KTP_UNKNOWN",1)
AND
DB_KTP_FilledPipesUnknownBlend(_SmokingPipe,_Status)
AND
DB_KTP_StatusPouch(_StatusTurns,_PuffCount,_Status,_TobaccoPouchTemplate)
AND
IntegerSubtract(_PuffCount,1,_NewPuffCount)
AND
NOT HasActiveStatus(_Player,"KTP_OVERDOSE_5",1)
AND
QRY_KTP_CheckHalsinPipe(_Player,_SmokingPipe,_Status)
AND
QRY_KTP_CheckForOverdose(_Player)
THEN
//SetDualEntityEvent(_SmokingPipe,_SmokingPipe,"USE PIPE UNKNOWN BLEND");
PlayEffect(_Player,(EFFECTRESOURCE)VFX_Status_Tharchiate_False_Life_Apply_01_e34f6123-0933-e70d-339e-d5a8c3490cb0,"Chest_M",1.0);
PlaySound(_Player, "GOB_GoblinKing_Scrying_SmokeStart");
ApplyStatus(_Player,_Status,_StatusTurns,1);
DB_KTP_FilledPipes(_SmokingPipe,_Status,_NewPuffCount,_StatusTurns); // Adding to DB with pipes with known blends
PROC_StudyIngredientsOfTobaccoBlend(_Status,(ITEMROOT)_TobaccoPouchTemplate); // study recipe 
NOT DB_KTP_FilledPipesUnknownBlend(_SmokingPipe,_Status); // Remove from DB with pipes with unknown blends
//
//
// -----------------------------------------------------------------------------------------
// SMOKING PIPE FILLING PROCESSING
// -----------------------------------------------------------------------------------------
//
// If an unknown blend is filled
IF 
DB_KTP_FilledPipesUnknownBlend(_NewSmokingPipe,_PouchStatus)
//AND
//GetHostCharacter(_Player)
THEN
//SetDualEntityEvent(_Player,_Player,"DB_KTP_FilledPipesUnknownBlend");
ApplyStatus(_NewSmokingPipe,"KTP_UNKNOWN",-1.0,1);
ApplyStatus(_NewSmokingPipe,"KTP_PUFF_COUNT_1",-1.0,1);
//
// If a known blend is filled
IF 
DB_KTP_FilledPipes(_NewSmokingPipe,_NewStatus,_NewPuffCount,_NewStatusTurns)
AND
_NewPuffCount > 0
AND
ConcatenateInteger("KTP_PUFF_COUNT_",_NewPuffCount,_NewStatusPuffCount) // new status for pipe
AND
Concatenate("ApplyStatus ",_NewStatusPuffCount,_StringDebug)
THEN
//SetDualEntityEvent(_NewSmokingPipe,_NewSmokingPipe,_StringDebug);
ApplyStatus(_NewSmokingPipe,_NewStatus,-1.0,1);
ApplyStatus(_NewSmokingPipe,_NewStatusPuffCount,-1.0,1); // PUFF COUNT
//
// SINGLETON FOR SMOKINGPIPE BY PUFFCOUNT
IF 
DB_KTP_FilledPipes(_NewSmokingPipe,_NewStatus,_NewPuffCount,_NewStatusTurns)
AND
DB_KTP_FilledPipes(_OldSmokingPipe,_OldStatus,_OldPuffCount,_OldStatusTurns)
AND
_NewSmokingPipe == _OldSmokingPipe
AND
_NewStatus == _OldStatus
AND
_NewStatusTurns == _OldStatusTurns
AND
_NewPuffCount != _OldPuffCount
AND
_NewPuffCount > 0
AND
ConcatenateInteger("KTP_PUFF_COUNT_",_NewPuffCount,_NewStatusPuffCount) // new status for pipe
THEN
//SetDualEntityEvent(_NewSmokingPipe,_NewSmokingPipe,"SINGLETON FOR PUFF COUNT");
ApplyStatus(_NewSmokingPipe,_NewStatus,-1.0,1);
ApplyStatus(_NewSmokingPipe,_NewStatusPuffCount,-1.0,1);
NOT DB_KTP_FilledPipes(_OldSmokingPipe,_OldStatus,_OldPuffCount,_OldStatusTurns);
//
// SINGLETON FOR SMOKINGPIPE BY STATUS
IF 
DB_KTP_FilledPipes(_NewSmokingPipe,_NewStatus,_NewPuffCount,_NewStatusTurns)
AND
DB_KTP_FilledPipes(_OldSmokingPipe,_OldStatus,_OldPuffCount,_OldStatusTurns)
AND
_NewSmokingPipe == _OldSmokingPipe
AND
_NewStatus != _OldStatus
AND
_NewPuffCount > 0
AND
ConcatenateInteger("KTP_PUFF_COUNT_",_NewPuffCount,_NewStatusPuffCount)
THEN
//SetDualEntityEvent(_NewSmokingPipe,_NewSmokingPipe,"SINGLETON BY STATUS");
ApplyStatus(_NewSmokingPipe,_NewStatus,-1.0,1);
ApplyStatus(_NewSmokingPipe,_NewStatusPuffCount,-1.0,1);
NOT DB_KTP_FilledPipes(_OldSmokingPipe,_OldStatus,_OldPuffCount,_OldStatusTurns);
//
// CLEAR PIPE IF PUFFCOUNT 0
IF 
DB_KTP_FilledPipes(_NewSmokingPipe,_NewStatus,_NewPuffCount,_NewStatusTurns)
AND
DB_KTP_FilledPipes(_OldSmokingPipe,_OldStatus,_OldPuffCount,_OldStatusTurns)
AND
_NewSmokingPipe == _OldSmokingPipe
AND
_NewStatus == _OldStatus
AND
_NewStatusTurns == _OldStatusTurns
AND
_NewPuffCount != _OldPuffCount
AND
_NewPuffCount <= 0
THEN
//SetDualEntityEvent(_NewSmokingPipe,_NewSmokingPipe,"CLEAR PIPE IF PUFFCOUNT 0");
RemoveStatus(_OldSmokingPipe,"KTP_PUFF_COUNT_1");
RemoveStatus(_OldSmokingPipe,"KTP_PUFF_COUNT_2");
RemoveStatus(_OldSmokingPipe,"KTP_PUFF_COUNT_3");
RemoveStatus(_OldSmokingPipe,"KTP_PUFF_COUNT_4");
RemoveStatus(_OldSmokingPipe,"KTP_PUFF_COUNT_5");
RemoveStatus(_OldSmokingPipe,_NewStatus);
RemoveStatus(_OldSmokingPipe,_OldStatus);
ClearTag(_NewSmokingPipe,KTP_FILLED_d0575f41-9d81-4f26-9f34-530c2162c110); //disables use action
NOT DB_KTP_FilledPipes(_NewSmokingPipe,_NewStatus,_NewPuffCount,_NewStatusTurns);
NOT DB_KTP_FilledPipes(_OldSmokingPipe,_OldStatus,_OldPuffCount,_OldStatusTurns);
//
//
// -----------------------------------------------------------------------------------------
// HALSIN PIPE PROCESSING
// -----------------------------------------------------------------------------------------
//
// (TAG)REALLY_HALSIN_9b8709f9-8d2a-4b4e-a465-8505c561d7f5
// FOR DEBUG (TAG)REALLY_SHADOWHEART_642d2aee-e3df-47e3-9f47-bbcd441bb9e0
//
// Halsin smoke
QRY
QRY_KTP_CheckHalsinPipe((CHARACTER)_Player,(ITEM)_SmokingPipe,(STRING)_Status)
AND
DB_KTP_HalsinsPipe(_SmokingPipe)
AND
NOT DB_KTP_StatusBad(_Status)
AND
//IsTagged(_Player,(TAG)REALLY_SHADOWHEART_642d2aee-e3df-47e3-9f47-bbcd441bb9e0,1)
IsTagged(_Player,(TAG)REALLY_HALSIN_9b8709f9-8d2a-4b4e-a465-8505c561d7f5,1)
THEN
ApplyStatus(_Player,"KTP_HALSIN_HAPPY",120.0,1);
//
// Halsins partner smoke
QRY
QRY_KTP_CheckHalsinPipe((CHARACTER)_Player,(ITEM)_SmokingPipe,(STRING)_Status)
AND
DB_KTP_HalsinsPipe(_SmokingPipe)
AND
NOT DB_KTP_StatusBad(_Status)
AND
//NOT IsTagged(_Player,(TAG)REALLY_SHADOWHEART_642d2aee-e3df-47e3-9f47-bbcd441bb9e0,1)
NOT IsTagged(_Player,(TAG)REALLY_HALSIN_9b8709f9-8d2a-4b4e-a465-8505c561d7f5,1)
AND
DB_ORI_Partnered(_Player,_Partner)
AND
//IsTagged(_Partner,(TAG)REALLY_SHADOWHEART_642d2aee-e3df-47e3-9f47-bbcd441bb9e0,1)
IsTagged(_Player,(TAG)REALLY_HALSIN_9b8709f9-8d2a-4b4e-a465-8505c561d7f5,1)
THEN
ApplyStatus(_Player,"KTP_HALSIN_PARTNER_HAPPY",60.0,1);
//
// QRY Must always be successful
QRY
QRY_KTP_CheckHalsinPipe((CHARACTER)_Player,(ITEM)_SmokingPipe,(STRING)_Status)
THEN
DB_NOOP(1);
//
//
// -----------------------------------------------------------------------------------------
// STUDE NEW BLEND PROCESSING
// -----------------------------------------------------------------------------------------
//
// Add to DB for known blends
PROC
PROC_StudyIngredientsOfTobaccoBlend((STRING)_Status,(ITEMROOT)_TobaccoPouchTemplate)
//AND
//GetHostCharacter(_Player)
AND
NOT DB_KTP_Known_Blend(_Status,_)
THEN
//SetDualEntityEvent(_Player,_Player,"PROC_StudyIngredientsOfTobaccoBlend");
DB_KTP_Known_Blend(_Status,_TobaccoPouchTemplate);
//
// PROCESS NEW KNOWN BLEND
//
// For all party inventory
// Check new known pouchs with unknown status in inventory in DB_KTP_UnknownBlendPouch
// We need to delete it and add new pouch with new known blend
IF
DB_KTP_Known_Blend(_StatusPattern,_TobaccoPouchTemplate)
AND
GetHostCharacter(_Player)
AND
DB_KTP_UnknownBlendPouch(_TobaccoPouchUnknown,(ITEMROOT)_TobaccoPouchTemplate,_Status)
AND
ItemIsInPartyInventory((ITEM)_TobaccoPouchUnknown, _Player, 0, 1)
THEN
//SetDualEntityEvent(_Player,_Player,"DB_KTP_Known_Blend Check new known pouchs ");
RequestDelete(_TobaccoPouchUnknown);
TemplateAddTo((ITEMROOT)_TobaccoPouchTemplate,_Player,1,0);
NOT DB_KTP_UnknownBlendPouch(_TobaccoPouchUnknown,_TobaccoPouchTemplate,_Status);
//
// For all party inventory
// Check new known smoking pipe in inventory DB_KTP_FilledPipesUnknownBlend
// We need to remove unknown status and apply new known status
IF
DB_KTP_Known_Blend(_Status,_TobaccoPouchTemplate)
AND
GetHostCharacter(_Player)
AND
DB_KTP_FilledPipesUnknownBlend(_SmokingPipe,_Status)
AND
NOT DB_KTP_FilledPipes(_SmokingPipe,_Status,_,_)
AND
ItemIsInPartyInventory((ITEM)_SmokingPipe, _Player, 0, 1)
AND
DB_KTP_StatusPouch(_StatusTurns,_PuffCount,_Status,_)
AND
ConcatenateInteger("KTP_PUFF_COUNT_",_PuffCount,_NewStatusPuffCount)
THEN
//SetDualEntityEvent(_Player,_Player,"DB_KTP_Known_Blend Check new known smoking pipe");
RemoveStatus(_SmokingPipe,"KTP_UNKNOWN");
//ApplyStatus(_SmokingPipe,_Status,-1.0,1);
//ApplyStatus(_SmokingPipe,_NewStatusPuffCount,-1.0,1);
DB_KTP_FilledPipes(_SmokingPipe,_Status,_PuffCount,_StatusTurns);
NOT DB_KTP_FilledPipesUnknownBlend(_SmokingPipe,_Status);
//
// PROCESSING ADDING ITEMS IN INVENTORY
// Pouch
IF
AddedTo(_TobaccoPouchUnknown, _Player, _)
AND
SysCount("DB_KTP_CurrentMortar",3,_Count) // Checking that the pouch is not added during the mixing process in the mortar
AND
DB_KTP_UnknownBlendPouch((ITEM)_TobaccoPouchUnknown,(ITEMROOT)_TobaccoPouchTemplate,_Status)
AND
DB_KTP_Known_Blend(_StatusPattern,_TobaccoPouchTemplate)
THEN
//SetDualEntityEvent(_Player,_Player,"PROCESSING ADDING ITEMS IN INVENTORY Pouch");
RequestDelete(_TobaccoPouchUnknown);
TemplateAddTo((ITEMROOT)_TobaccoPouchTemplate,_Player,1,0);
NOT DB_KTP_UnknownBlendPouch(_TobaccoPouchUnknown,_TobaccoPouchTemplate,_Status);
//
// Smoking pipe
IF
AddedTo(_SmokingPipe, _Player, _)
AND
DB_KTP_FilledPipesUnknownBlend((ITEM)_SmokingPipe,_Status)
AND
NOT DB_KTP_FilledPipes(_SmokingPipe,_Status,_,_)
AND
DB_KTP_Known_Blend(_Status,_)
AND
DB_KTP_StatusPouch(_StatusTurns,_PuffCount,_Status,_)
AND
ConcatenateInteger("KTP_PUFF_COUNT_",_PuffCount,_NewStatusPuffCount)
THEN
//SetDualEntityEvent(_Player,_Player,"PROCESSING ADDING ITEMS IN INVENTORY _SmokingPipe");
RemoveStatus(_SmokingPipe,"KTP_UNKNOWN");
//ApplyStatus(_SmokingPipe,_Status,-1.0,1);
//ApplyStatus(_SmokingPipe,_NewStatusPuffCount,-1.0,1);
DB_KTP_FilledPipes(_SmokingPipe,_Status,_PuffCount,_StatusTurns);
NOT DB_KTP_FilledPipesUnknownBlend(_SmokingPipe,_Status);
//
//
// -----------------------------------------------------------------------------------------
// OVERDOSE PROCESSING
// -----------------------------------------------------------------------------------------
//
// Five stages of overdose. The third is where problems begin, the fifth is nicotine poisoning.
//
// MAX KTP_OVERDOSE_5 - cannot smoke
IF
UseFinished(_Player,_SmokingPipe,1)
AND
DB_KTP_FilledPipes(_SmokingPipe,_Status,_PuffCount,_StatusTurns)
AND
_PuffCount > 0
AND
HasActiveStatus(_Player,"KTP_OVERDOSE_5",1)
THEN
PROC_PlayCantUseItemAD((CHARACTER)_Player);
//
// Exist overdose
QRY
QRY_KTP_CheckForOverdose((CHARACTER)_Player)
AND
DB_KTP_Overdose(_Player,_Count)
AND
_Count < 5
AND
IntegerSum(_Count,1,_NewCount)
//AND
//ConcatenateInteger("QRY_KTP_CheckForOverdose EXIST ",_NewCount,_StringDebug)
THEN
//SetDualEntityEvent(_Player,_Player,_StringDebug);
DB_KTP_Overdose(_Player,_NewCount);
//
// IF Count 3 we need apply status
QRY
QRY_KTP_CheckForOverdose((CHARACTER)_Player)
AND
DB_KTP_Overdose(_Player,_Count)
AND
_Count == 3
THEN
//SetDualEntityEvent(_Player,_Player,"ApplyStatus KTP_OVERDOSE_3");
ApplyStatus(_Player,"KTP_OVERDOSE_3",30.0,1);
//
// IF Count 4 we need apply status
QRY
QRY_KTP_CheckForOverdose((CHARACTER)_Player)
AND
DB_KTP_Overdose(_Player,_Count)
AND
_Count == 4
THEN
//SetDualEntityEvent(_Player,_Player,"ApplyStatus KTP_OVERDOSE_4");
ApplyStatus(_Player,"KTP_OVERDOSE_4",90.0,1);
//
// IF Count 5 we need apply status
QRY
QRY_KTP_CheckForOverdose((CHARACTER)_Player)
AND
DB_KTP_Overdose(_Player,_Count)
AND
_Count == 5
THEN
//SetDualEntityEvent(_Player,_Player,"ApplyStatus KTP_OVERDOSE_5");
ApplyStatus(_Player,"KTP_OVERDOSE_5",-1.0,1);
//
// New Overdose
QRY
QRY_KTP_CheckForOverdose((CHARACTER)_Player)
AND
NOT DB_KTP_Overdose(_Player,_)
THEN
//SetDualEntityEvent(_Player,_Player,"QRY_KTP_CheckForOverdose NEW");
DB_KTP_Overdose(_Player,1);
//
// QRY Must always be successful
QRY
QRY_KTP_CheckForOverdose((CHARACTER)_Player)
THEN
DB_NOOP(1);
//
// After a long rest, we clear the overdose DB
PROC
PROC_LongRest()
THEN
SysClear("DB_KTP_Overdose",2);
//
// If the status is removed from the player without clearing the DB, then we write the overdose level in the DB to 1 less than it was
IF
StatusRemoved(_Player,_StatusOverdose,_Cause,_)
AND
_StatusOverdose == "KTP_OVERDOSE_5"
AND
DB_KTP_Overdose((CHARACTER)_Player,_)
THEN
DB_KTP_Overdose((CHARACTER)_Player,4);
//
// Applied if not Saving Throw for Ability.Constitution in statuses KTP_OVERDOSE_*
// Then we need remove all Tobacco Statuses
IF
StatusApplied(_Player,"KTP_REMOVE_STATUSES",_,_)
AND
DB_KTP_StatusPouch(_,_,_TobaccoStatus,_)
AND
HasActiveStatus(_Player,_TobaccoStatus,1)
THEN
RemoveStatus(_Player,_TobaccoStatus);
//
// Applied if used ALCH_POTION_REST_SLEEP_GREATER_RESTORATION
// Then we need remove all Tobacco Statuses and clear the overdose DB
IF
StatusApplied(_Player,"ALCH_POTION_REST_SLEEP_GREATER_RESTORATION",_,_)
AND
DB_KTP_StatusPouch(_,_,_TobaccoStatus,_)
AND
HasActiveStatus(_Player,_TobaccoStatus,1)
THEN
RemoveStatus(_Player,_TobaccoStatus);
//
IF
StatusApplied(_Player,"ALCH_POTION_REST_SLEEP_GREATER_RESTORATION",_,_)
THEN
RemoveStatus(_Player,"KTP_OVERDOSE_3");
RemoveStatus(_Player,"KTP_OVERDOSE_4");
RemoveStatus(_Player,"KTP_OVERDOSE_5");
SysClear("DB_KTP_Overdose",2);
//
// Singleton for DB_KTP_Overdose
IF
DB_KTP_Overdose(_NewPlayer,_NewCount)
AND
DB_KTP_Overdose(_OldPlayer,_OldCount)
AND
_NewPlayer == _OldPlayer
AND
_NewCount != _OldCount
THEN
NOT DB_KTP_Overdose(_OldPlayer,_OldCount);
